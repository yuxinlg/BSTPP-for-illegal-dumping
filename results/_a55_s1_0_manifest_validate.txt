"""Phase 3f completion-manifest validator.

Same assertion set the A-54 integration and its follow-up ran (recorded in
refactor-patches/phase3f/rebaseline_record_a54.md section 3): schema_version,
the immutable evidence-baseline commit, next_free_ids, the WP1-WP10 set, the
responsibility_coverage mapping onto declared packages, the five vertical
slices, and corrective_cycles_used present on every package. Prints
MANIFEST_OK <n_work_packages> <n_responsibility_items> and its own exit status.
"""
import sys

import yaml

PATH = "docs/phase3f_completion_manifest.yaml"
EXPECTED_WPS = {f"WP{i}" for i in range(1, 11)}
EXPECTED_SLICES = {
    "S0_operational_reset", "S1_configuration_spine", "S2_runtime_ownership",
    "S3_artifacts_and_interfaces", "S4_architectural_cutover",
}


def main() -> int:
    print(f"interpreter : {sys.executable}")
    print(f"parser      : PyYAML {yaml.__version__}")
    with open(PATH, encoding="utf-8") as fh:
        d = yaml.safe_load(fh)

    fails = []

    def check(cond, msg):
        if not cond:
            fails.append(msg)

    check(d["schema_version"] == 1, "schema_version != 1")
    check(d["as_of"]["commit_short"] == "c4e069f",
          "as_of.commit_short moved off the immutable evidence baseline")
    check(d["as_of"]["adopted_at_commit_short"] == "6072a19",
          "as_of.adopted_at_commit_short moved")
    nf = d["as_of"]["next_free_ids"]
    check(nf == {"amendment": "A-55", "decision": "D-63",
                 "config_invariant": "CI-11", "open_item": "OP-32"},
          f"next_free_ids moved: {nf}")

    wps = d["work_packages"]
    check(set(wps) == EXPECTED_WPS, f"work packages are not WP1-WP10: {sorted(wps)}")
    for name, wp in sorted(wps.items()):
        check("corrective_cycles_used" in wp,
              f"{name} has no corrective_cycles_used")
        check("scope" in wp, f"{name} has no scope block")

    slices = d["vertical_slices"]
    check(set(slices) == EXPECTED_SLICES, f"vertical_slices moved: {sorted(slices)}")

    items = d["responsibility_coverage"]["items"]
    check(bool(items), "responsibility_coverage.items is empty")
    for k, v in items.items():
        check(v in wps, f"responsibility '{k}' names undeclared owner {v!r}")

    # Every declared scope.files entry must exist in the tree.
    import os
    for name, wp in sorted(wps.items()):
        for f in wp.get("scope", {}).get("files", []) or []:
            check(os.path.isfile(f), f"{name} scope.files names missing {f}")

    if fails:
        for f in fails:
            print(f"FAIL: {f}")
        return 1

    print(f"MANIFEST_OK {len(wps)} {len(items)}")
    s1 = slices["S1_configuration_spine"]
    print(f"S1 status: {s1['status']}  scope_declared: {s1['scope_declared']}"
          f"  closes_under: {s1['closes_under_amendment']}")
    print(f"S1 order: {len(s1['order'])} commits; decisions: "
          f"{len(s1['decisions'])}; exclusions: {len(s1['exclusions'])}")
    print("WP1 scope.files: " + ", ".join(wps["WP1"]["scope"]["files"]))
    print(f"WP9 intake rows: {len(wps['WP9']['intake'])} -> "
          + ", ".join(r["id"] for r in wps["WP9"]["intake"]))
    print("corrective_cycles_used: " + ", ".join(
        f"{k}={v['corrective_cycles_used']}" for k, v in sorted(
            wps.items(), key=lambda kv: int(kv[0][2:]))))
    print(f"next_free_ids UNCHANGED: {nf}")
    return 0


if __name__ == "__main__":
    rc = main()
    print(f"EXIT_STATUS:{rc}")
    sys.exit(rc)
